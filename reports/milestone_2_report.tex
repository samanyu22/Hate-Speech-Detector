\section{Dataset Description and Statistics}
Conventional hate speech models trained solely on high-resource English collapse when encountering Indic dialects, code-mixed Latin transliteration (Hinglish), and multimodal memes. To ensure robust real-world generalization, we curated a multi-corpus benchmark totaling \textbf{1,310 authenticated instances} across four foundational sources:
\begin{itemize}
    \item \textbf{HASOC Hi-En (840):} English, native Hindi (Devanagari), and code-mixed Hinglish capturing informal abusive slang and civil debate.
    \item \textbf{HateXplain Benchmark (200):} Annotated with ground-truth token-level rationales for XAI faithfulness auditing.
    \item \textbf{Subtle \& Structural Benchmark (240):} Veiled stereotypes, institutional discrimination, and counter-speech negations.
    \item \textbf{Multimodal Hateful Memes (30):} Social media memes with typography extracted via EasyOCR.
\end{itemize}

\textbf{Class Distribution \& Splitting:} Strict binary classification ($y \in \{0, 1\}$), where $0$ is Non-Hateful and $1$ is Hateful. Partitions use a leakage-safe stratified 70/15/15 split ($S=42$):

\section{Methodology and Implementation}
\textbf{Mathematical Formulation:} Social media inputs are modeled as multi-component tuples:
\begin{equation}
    x_i = \{T_i, V_i, C_i\} \implies f_\theta(x_i) \to y_i \in \{0, 1\}
\end{equation}
where $T_i$ denotes text, $V_i$ represents the visual image tensor, and $C_i$ denotes sanitized conversational context.

\textbf{Operational Architecture:}
\begin{enumerate}
    \item \textbf{Preprocessing ($T_i$):} Unicode NFKC normalization, URL neutralization, Devanagari script detection ($U+0900$--$U+097F$), and phonetic Hinglish elongation reduction.
    \item \textbf{Vision \& OCR ($V_i$):} Bilinear resizing ($224 \times 224$) and EasyOCR extraction yielding bounding boxes $[y_{\min}, x_{\min}, y_{\max}, x_{\max}]$ and typography text.
    \item \textbf{Backbones \& Fusion:} Classical Scikit-Learn TF-IDF baseline alongside a PyTorch Multilingual Transformer with multi-head self-attention. Visual CNN embeddings ($128$-d) and text tokens are fused via intermediate cross-modal cross-attention with sigmoid gating:
    \begin{equation}
        h_{\text{fused}} = g \odot h_{\text{text}} + (1 - g) \odot \text{CrossAttn}(h_{\text{text}}, h_{\text{vis}})
    \end{equation}
    \item \textbf{Bidirectional Local Negation Scope (Way 1):} A $[-2, +2]$ window decoupling negated slurs from zero-tolerance hate escalation for English prefixes (\textit{"not vermin"}) and Hindi suffixes (\textit{"harami nahi"}).
    \item \textbf{Grounded XAI:} Leave-One-Out perturbation $\Delta_j = f(x) - f(x_{\backslash j})$, formally verified via AOPC faithfulness.
    \item \textbf{Reward Optimization (DPO):} Multi-objective reward:
    \begin{equation}
        R = \alpha R_{\text{cls}} + \beta R_{\text{exp}} + \gamma R_{\text{ctx}} + \delta R_{\text{fb}} - \lambda_{\text{err}} R_{\text{err}}
    \end{equation}
    ($\alpha=0.35, \beta=0.25, \gamma=0.15, \delta=0.25, \lambda_{\text{err}}=0.50$). User preference pairs $(x, y_w, y_l)$ update weights via Direct Preference Optimization.
\end{enumerate}

\textbf{Key Findings:}
\begin{itemize}
    \item \textbf{Target Grounding Disparity:} Classical TF-IDF flags benign insults (\textit{"you are absolutely stupid..."}) as Hateful ($51.74\%$). The Transformer outputs Non-Hateful ($99.5\%$ confidence) due to absence of demographic target grounding.
    \item \textbf{Slur Rebuttal Resolution:} Way 1 Bidirectional Negation lifted negated counter-speech from 65\% to \textbf{100\% (20/20)} with zero false positives.
    \item \textbf{Multimodal Synergy:} Cross-attention yielded \textbf{+5.0\% F1} on memes with text-image semantic contradictions.
    \item \textbf{Test Suite:} \textbf{45 / 45 automated pytest tests pass}.
\end{itemize}

\section{Challenges Encountered}
\begin{enumerate}
    \item \textbf{Code-Mixed Hinglish Tokenization Collapse:} Standard tokenizers fracture Romanized Hindi into unknown characters. \textit{Resolved} via script distribution analysis, phonetic Hinglish dictionaries, and character n-gram union.
    \item \textbf{Slur Rebuttal Blindspot:} Slur blacklists flagged valid defense (\textit{"immigrants are not vermin"}). \textit{Resolved} via $[-2, +2]$ bidirectional local negation scoping.
    \item \textbf{Double Inversions:} Counter-speech guards bypassed by inverted condemnation (\textit{"will not condemn attacks... deserved it"}). \textit{Resolved} via condemnation negation checks and subordination escalation.
    \item \textbf{Cross-Machine Sync:} Missing PyTorch weights on git pull. \textit{Resolved} via `.gitignore` whitelisting and dynamic relative path sanitization.
\end{enumerate}

\section{Future Work (Before Final Evaluation)}
